cmake_minimum_required(VERSION 3.24)
project(SPIKES_Studio_Runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenMP REQUIRED)
if(NOT EXISTS "${EIGEN_INCLUDE}/Eigen/Core")
  message(FATAL_ERROR "Provide EIGEN_INCLUDE pointing to Eigen headers")
endif()
set(SRC "${CMAKE_CURRENT_LIST_DIR}/../../src/spikes")
if(NOT EXISTS "${SRC}/c_api.cpp")
  set(SRC "${CMAKE_CURRENT_LIST_DIR}/../../engine/src/spikes")
endif()
add_library(spikes_c_api SHARED ${SRC}/dc_solver.cpp ${SRC}/osdi_device.cpp ${SRC}/transient_solver.cpp ${SRC}/transient_session.cpp ${SRC}/c_api.cpp)
target_include_directories(spikes_c_api PRIVATE "${SRC}/.." "${EIGEN_INCLUDE}")
target_link_libraries(spikes_c_api PRIVATE OpenMP::OpenMP_CXX ${CMAKE_DL_LIBS})
target_compile_definitions(spikes_c_api PRIVATE SPIKES_SHARED SPIKES_BUILDING_LIBRARY)
if(NOT MSVC)
  target_compile_options(spikes_c_api PRIVATE -fno-fast-math -ffp-contract=off)
endif()
