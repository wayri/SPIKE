cmake_minimum_required(VERSION 3.20)
project(SPIKENativeAirflow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(spike_airflow STATIC cavity.cpp)
target_include_directories(spike_airflow PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(spike_airflow PRIVATE /W4 /fp:strict)
else()
  target_compile_options(spike_airflow PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
endif()
add_executable(spike_native_airflow main.cpp)
target_link_libraries(spike_native_airflow PRIVATE spike_airflow)
enable_testing()
add_executable(test_native_airflow "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_native_airflow.cpp")
target_link_libraries(test_native_airflow PRIVATE spike_airflow)
add_test(NAME native_airflow COMMAND test_native_airflow)
target_sources(spike_airflow PRIVATE cht.cpp)
add_executable(spike_native_cht cht_main.cpp)
target_link_libraries(spike_native_cht PRIVATE spike_airflow)
add_executable(test_native_cht "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_native_cht.cpp")
target_link_libraries(test_native_cht PRIVATE spike_airflow)
add_test(NAME native_cht COMMAND test_native_cht)
